import java.io.BufferedReader;
import java.io.FileReader;
import java.io.IOException;
import java.util.*;

public class WordCount {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        String letter = "";
        while (!letter.equalsIgnoreCase("quit")) {
            System.out.print("Enter a letter to search for or 'quit' to exit: ");
            letter = scanner.nextLine();
            if (letter.equalsIgnoreCase("quit")) {
                break;
            }
            String fileName = "input.txt"; // the name of the text file to read
            Map<String, Integer> wordCounts = new TreeMap<>(); // a map to store word counts

            try (BufferedReader reader = new BufferedReader(new FileReader(fileName))) {
                String line;
                while ((line = reader.readLine()) != null) {
                    line = line.replaceAll("\\p{Punct}", ""); // remove all punctuation marks
                    String[] words = line.split("\\s+"); // split the line into words
                    for (String word : words) {
                        if (word.toLowerCase().startsWith(letter.toLowerCase())) { // check if the word starts with the given letter
                            wordCounts.put(word.toLowerCase(), wordCounts.getOrDefault(word.toLowerCase(), 0) + 1); // update word counts
                        }
                    }
                }
            } catch (IOException e) {
                System.err.println("Error reading file: " + e.getMessage());
                System.exit(1);
            }

            if (wordCounts.isEmpty()) {
                System.out.println("No words found starting with the letter " + letter);
            } else {
                int totalCount = 0;
                System.out.println("Words starting with " + letter.toUpperCase() + ":");
                for (Map.Entry<String, Integer> entry : wordCounts.entrySet()) {
                    String word = entry.getKey();
                    int count = entry.getValue();
                    System.out.println(word + " - " + count);
                    totalCount += count;
                }
                System.out.println("Total count: " + totalCount);
            }
        }
    }
}
